const SHEET_NAME = "PDF BESOK"; function doGet() { try { const ss = SpreadsheetApp.openById(SpreadsheetApp.getActiveSpreadsheet().getId()); const sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { return ContentService.createTextOutput("SHEET_NOT_FOUND"); } const data = sheet.getDataRange().getValues(); return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput("ERROR: " + err); } }
