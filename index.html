function loadData(){ fetch(API_URL + "?t=" + new Date().getTime()) .then(r => r.text()) .then(text => { let data = JSON.parse(text); let html = ""; let total = 0; let tanggal = ""; data.forEach(row => { const isi = row[0]; const tgl = row[1]; const status = row[2] || ""; if(!isi) return; total++; if(!tanggal && tgl){ tanggal = new Date(tgl).toLocaleDateString("id-ID",{ day:"2-digit", month:"long", year:"numeric" }); } const sudah = status === "TERKIRIM"; /* 🔥 AMBIL WA DARI TEKS ORDER (ISI) */ let wa1 = ""; let wa2 = ""; const wa1Match = isi.match(/WA\s*1\s*[:\-]?\s*(08\d{8,13})/i); const wa2Match = isi.match(/WA\s*2\s*[:\-]?\s*(08\d{8,13})/i); if(wa1Match) wa1 = wa1Match[1]; if(wa2Match) wa2 = wa2Match[1]; let waButtons = ""; if(wa1){ waButtons += `WA 1`; } if(wa2){ waButtons += `WA 2`; } html += `
${isi}
${waButtons}
`; }); document.getElementById("jumlah").innerHTML = `
📅 ${tanggal}
Total Pengiriman: ${total}
`; document.getElementById("list").innerHTML = html; }) .catch(err => { console.log(err); document.getElementById("list").innerHTML = "❌ Gagal load data"; }); }
